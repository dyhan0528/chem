// 화학정복하기 진도 저장 API (Vercel 서버리스 함수) // 저장소: Upstash Redis (Vercel Storage에서 연결하면 환경 변수가 자동으로 들어옵니다) // GET /api/progress → 전체 진도 읽기 // POST /api/progress {id,value} → 강의 하나 저장 // DELETE /api/progress → 전체 초기화 // 모든 요청은 x-pass 헤더에 비밀번호가 있어야 합니다. const KEY = "doyoon:chem:progress"; const PASSWORD = process.env.SITE_PASSWORD || "1234"; const URL_ = process.env.KV_REST_API_URL || process.env.UPSTASH_REDIS_REST_URL; const TOKEN = process.env.KV_REST_API_TOKEN || process.env.UPSTASH_REDIS_REST_TOKEN; async function redis(command) { const res = await fetch(URL_, { method: "POST", headers: { Authorization: `Bearer ${TOKEN}`, "Content-Type": "application/json" }, body: JSON.stringify(command), }); const data = await res.json(); if (!res.ok || data.error) throw new Error(data.error || `redis ${res.status}`); return data.result; } function readBody(req) { if (req.body && typeof req.body === "object") return Promise.resolve(req.body); return new Promise((resolve) => { let raw = ""; req.on("data", (c) => (raw += c)); req.on("end", () => { try { resolve(JSON.parse(raw || "{}")); } catch { resolve({}); } }); }); } module.exports = async (req, res) => { res.setHeader("Cache-Control", "no-store"); if ((req.headers["x-pass"] || "") !== PASSWORD) { return res.status(401).json({ error: "비밀번호가 맞지 않아요" }); } if (!URL_ || !TOKEN) { return res.status(503).json({ error: "저장소가 연결되지 않았어요 (Vercel → Storage에서 Upstash Redis 연결)" }); } try { if (req.method === "GET") { const flat = (await redis(["HGETALL", KEY])) || []; const progress = {}; for (let i = 0; i < flat.length; i += 2) { try { progress[flat[i]] = JSON.parse(flat[i + 1]); } catch {} } return res.status(200).json({ progress }); } if (req.method === "POST") { const { id, value } = await readBody(req); if (!/^L\d{2}$/.test(id || "") || typeof value !== "object" || value === null) { return res.status(400).json({ error: "잘못된 요청이에요" }); } const clean = { done: !!value.done, date: String(value.date || "").slice(0, 10), score: String(value.score || "").slice(0, 10), }; await redis(["HSET", KEY, id, JSON.stringify(clean)]); return res.status(200).json({ ok: true }); } if (req.method === "DELETE") { await redis(["DEL", KEY]); return res.status(200).json({ ok: true }); } res.setHeader("Allow", "GET, POST, DELETE"); return res.status(405).json({ error: "지원하지 않는 요청이에요" }); } catch (e) { return res.status(500).json({ error: "저장소 오류", detail: String(e.message || e) }); } }; # 한도윤 화학정복하기 장풍 선생님 화학(31강 + 추가 강의) 진도표. Vercel에 올려서 씁니다. ## 파일 - `index.html` — 화면 (비밀번호 화면, 진도 대시보드, 강의 체크·수강일·누테 점수) - `api/progress.js` — 진도를 클라우드(Upstash Redis)에 저장하는 서버 함수 ## 배포 순서 1. GitHub에서 새 저장소 `doyoon-chem` 만들기 → 이 폴더의 파일을 그대로 업로드 2. Vercel → Add New → Project → `doyoon-chem` 선택 → Deploy 3. Vercel 프로젝트 → Storage → Upstash (Redis) 만들기 → 이 프로젝트에 연결 4. Deployments → 최신 배포 오른쪽 ⋯ → Redeploy ## 비밀번호 바꾸기 (선택) Vercel → Settings → Environment Variables에 `SITE_PASSWORD`를 넣고 Redeploy. 기본값은 1234입니다. 바꿨다면 `index.html`의 `const PASSWORD = "1234";`도 같은 값으로 고치세요 (저장소 연결이 안 됐을 때만 쓰이는 값입니다).
